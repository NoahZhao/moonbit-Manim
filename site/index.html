  @mobject.ObjectState::shape(
    @geometry.square(2.0),
  ),
)

let circle = @mobject.ObjectState::shape(
  @geometry.circle(1.0),
)

scene.play(
  [@animation.transform(square, circle)],
  duration=2.0,
)
